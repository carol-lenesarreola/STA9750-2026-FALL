get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

Data Acquisition - Energy Consumption

Task 1b: Data Acquisition -NTD Energy Usage

Next, we download the 2025 Annual Database Energy Consumption report.

This dataset uses three key identifier columns: NTD ID, Mode, and Agency Name, followed by 2023 fuel consumption metrics. Combustion fuel columns are measured in gallons, while electric propulsion values are recorded in kWh.

The “Mode” column differentiates distinct transit types run by a single operator, such as distinguishing MTA buses from subways. However, these raw 2-letter transit codes require translation to be readable.

load_ntd_energy <- function(){
  # Offline link directed to the exact file manually downloaded
  NTD_ENERGY_FILE <- "data/mp02/2025_Energy_Consumption.xlsx"
  
  # Data processing framework 
  to_numeric_fill_0 <- function(x){
    x <- as.character(x)
    x <- if_else(x == "-", "0", x)
    val <- as.numeric(x)
    ifelse(is.na(val), 0, val)
    
  }
  
  df_clean <- read_excel(NTD_ENERGY_FILE, sheet = "Energy Consumption") |>
    select(-c(`Reporter Type`, `Reporting Module`, `Gallons of Other Fuel`, `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, `Mode`, `TOS`), to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarise(across(where(is.numeric), sum), .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
  
  return(df_clean)
}

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

Question 3: On average, how many pounds of CO2 are emitted per MWh of electricity produced in the US? (Inline answer). To ensure structural accuracy across states of vastly differing output sizes, we calculate a generation-weighted average of pounds of CO2 emitted per MWh nationally.

Code 
# To ensure this national indicator is accurately balanced, a basic unweighted average would fail because it treats small energy producers identically to massive industrial power grids like Texas or California. By implementing a generation-weighted average calculation Sum (CO2_MWh x generation_MWh) / Sum (generation_MWh), our formula scales each state's carbon footprint coefficient by its actual total productivity output. This provides a mathematically rigorous coefficient reflecting the true carbon footprint baseline of the unified American electrical framework.

national_weighted_co2 <- EIA_SEP_REPORT |>
  summarise(weighted_avg = sum(CO2_MWh * generation_MWh, na.rm = TRUE) / sum(generation_MWh, na.rm = TRUE)) |>
  pull(weighted_avg)

On average, the United States emits 784.86 pounds of CO2 per Megawatt-Hour (MWh) of electricity produced across all regional grid divisions.

emissions_avoided_table <- agency_efficiency |>
